                # --- Knowledge Base ---
                def get_response(message):
                    msg = message.lower()
                 
                    # Greetings
                    if any(word in msg for word in ["hi", "hello", "hey"]):
                        return "Hello there! What can I help you with today?"
                 
                    # How are you
                    elif "how are you" in msg:
                        return "I'm running great on Python! How about you?"
                 
                    # Name question
                    elif "your name" in msg:
                        return "I'm Nova, your AI learning assistant!"
                 
                    # Python questions
                    elif "python" in msg:
                        return ("Python is awesome! It's used at Google, NASA, and Instagram. "
                                "Want to learn it? Check out the 9 modules in PyKnowledge!")
                 
                    # Help topics
                    elif "modules" in msg:
                        return ("PyKnowledge has 9 CHED-aligned modules:\n"
                                "1. Introduction to Python\n"
                                "2. Control Structures\n"
                                "3. Loops\n"
                                "4. Functions\n"
                                "5. Data Structures\n"
                                "6. Modules and Imports\n"
                                "7. Input and Output\n"
                                "8. Errors and Exceptions\n"
                                "9. Classes and OOP")
                 
                    # Offline question
                    elif "offline" in msg:
                        return ("PyKnowledge works fully offline!\n"
                                "• All lessons cached on device\n"
                                "• No internet needed for exercises\n"
                                "• Perfect for patchy signal areas")
                 
                    # Installation help
                    elif "install" in msg:
                        return ("To install PyKnowledge:\n"
                                "1. Tap 'Download App' on the landing page\n"
                                "2. Chrome: ⋮ menu → Add to Home Screen\n"
                                "3. iPhone: Share → Add to Home Screen")
                 
                    # Quiz info
                    elif "quiz" in msg:
                        return ("Quizzes appear at each module's end.\n"
                                "You need 70% to pass. Don't worry — "
                                "you can retry as many times as needed!")
                 
                    # Jokes for fun
                    elif "joke" in msg:
                        import random
                        jokes = [
                            "Why do Python programmers wear glasses? Because they can't C!",
                            "What's a snake's favorite programming language? Python!",
                            "Why was the function sad? It didn't get any calls back."
                        ]
                        return random.choice(jokes)
                 
                    # Math calculations
                    elif "calculate" in msg or "math" in msg:
                        return "I can help with math! Try: calculate 2 + 2"
                 
                    # Farewell
                    elif any(word in msg for word in ["bye", "goodbye", "quit"]):
                        return "Goodbye! Happy coding! 🐍"
                 
                    # Default response
                    else:
                        return ("I can help with:\n"
                                "• Modules & curriculum\n"
                                "• Python basics\n"
                                "• Offline features\n"
                                "• Installation guide\n"
                "• Quiz info\n"
                                "• Jokes! Just ask 'tell me a joke'")
                 
                 
                # --- Main Chat Loop ---
                def main():
                    print("=" * 45)
                    print("  NOVA AI CHATBOX - Your Python Learning Buddy")
                    print("=" * 45)
                    print("Type 'quit', 'bye', or 'exit' to end the chat.")
                    print("Try: hello, modules, python, offline, joke, quiz")
                    print("-" * 45)
                 
                    # Conversation history
                    history = []
                 
                    while True:
                        # Get user input
                        try:
                            user_input = input("\nYou: ").strip()
                        except EOFError:
                            break
                 
                        # Check for empty input
                        if not user_input:
                            print("Nova: Please type something!")
                            continue
                 
                        # Check for exit commands
                        if user_input.lower() in ["quit", "bye", "exit"]:
                            print(f"Nova: {get_response('bye')}")
                            print(f"\nChat ended. You had {len(history)} exchanges.")
                            break
                 
                        # Get bot response
                        response = get_response(user_input)
                 
                        # Display and store
                        print(f"Nova: {response}")
                        history.append({"user": user_input, "bot": response})
                 
                    return history
                 
                 
                # --- Run the program ---
                if __name__ == "__main__":
                    conversation = main()

          About
          
A free Python course that works where the signal doesn't

          
PyKnowledge is built for students in Tawi-Tawi and across BARMM — designed
             around real classroom constraints, not ideal conditions.

        

        

          

            
            
What it is

            
Nine CHED-aligned modules with lessons, exercises, and quizzes — from your
               first print() to classes and object-oriented programming. Free, no ads.

          

          

            
            
Built in Tawi-Tawi

            
The curriculum, pacing, and offline-first design all come from teaching TRAC
               and BARMM students on shared computers with patchy island signal.

          

          

            
            
How to use it

            
Open the app, create a local profile (no email needed), and follow the path.
               Install it to your home screen once and everything works offline after that.

          

        

        

          

            
Where this curriculum is taught

            
Lesson content follows the CHED calendar and BARMM education standards
               used by schools across the Tawi-Tawi islands.
